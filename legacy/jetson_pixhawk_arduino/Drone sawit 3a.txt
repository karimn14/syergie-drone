enhanced_route_generator = """
# enhanced_route_generator.py – Spline-smoothed route with altitude control

import json
import numpy as np
from scipy.interpolate import splprep, splev

def generate_smoothed_plan(waypoints, output_file="route_pollination_smoothed.plan", altitude=10):
    lats = [wp["lat"] for wp in waypoints]
    lons = [wp["lon"] for wp in waypoints]

    # Fit spline to coordinates
    tck, _ = splprep([lats, lons], s=0)
    u_fine = np.linspace(0, 1, len(waypoints) * 5)  # Denser points
    smoothed = splev(u_fine, tck)

    plan = {
        "fileType": "Plan",
        "mission": {
            "cruiseSpeed": 5, "firmwareType": 12, "hoverSpeed": 3,
            "items": [],
            "plannedHomePosition": [smoothed[0][0], smoothed[1][0], altitude],
            "vehicleType": 2, "version": 2
        },
        "geoFence": {"circles": [], "polygons": [], "version": 2},
        "rallyPoints": {"points": [], "version": 2},
        "version": 1
    }

    for i in range(len(smoothed[0])):
        plan["mission"]["items"].append({
            "command": 16, "doJumpId": i+1, "frame": 3, "type": "SimpleItem",
            "params": [0, 0, 0, None, smoothed[0][i], smoothed[1][i], altitude],
            "autoContinue": True, "Altitude": altitude, "AltitudeMode": 1
        })

    with open(output_file, "w") as f:
        json.dump(plan, f, indent=2)
"""

# vision_recheck.py – Optional visual confirmation before spray
# Requires a camera feed and preloaded classifier (e.g., YOLOv8 or CV2 template matching)

import cv2

# Load YOLOv8 model (or use a lightweight classifier)

def confirm_tree_visual(camera_index=0, timeout=3):
    cap = cv2.VideoCapture(camera_index)

    frames = 0

    while frames < timeout * 10:  # ~10 FPS

        if not ret:
            break
        results = model(frame)
        labels = results.pandas().xyxy[0]['name'].tolist()
        if 'female_bloom' in labels:
            confirmed = True
            break
        frames += 1

    cap.release()
    return confirmed

confidence_weighted_mapper = """
# spray_command_mapper_confidence.py – Adjust spray duration using AI confidence score

import pandas as pd

def build_spray_commands(trees, dosage_file="pollen_dosage_table.csv"):
    dosage_df = pd.read_csv(dosage_file)
    commands = []

    for t in trees:
        age = t.get("age", 7)
        confidence = t.get("confidence", 0.9)  # default if not given

        match = dosage_df.iloc[(dosage_df["Tree Age (Years)"] - age).abs().argsort()[:1]]
        base_duration = int(match["Spray Duration (ms)"].values[0])

        # Scale spray based on confidence (linear scaling 0.7–1.0)
        scale = min(max(confidence, 0.7), 1.0)
        duration = int(base_duration * scale)

        commands.append({
            "tree_id": t["id"],
            "lat": t["lat"],
            "lon": t["lon"],
            "confidence": confidence,
            "duration_ms": duration,
            "command": f"S,{duration}"
        })

    pd.DataFrame(commands).to_csv("spray_commands_confidence.csv", index=False)
    return commands
"""

pwm_sprayer_ino = """
// pwm_sprayer_control.ino – PWM-based spray actuation with feedback

#define SPRAY_PIN 9         // PWM pin
#define LED_PIN 13          // Status LED
#define FLOW_PIN A0         // Analog flow sensor
#define SPRAY_TIMEOUT 5000  // Safety timeout (ms)

unsigned long sprayStart = 0;
bool spraying = false;
int sprayCount = 0;

void setup() {
  pinMode(LED_PIN, OUTPUT);
  pinMode(SPRAY_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  Serial.begin(9600);
  while (!Serial);
  Serial.println("🟢 PWM Sprayer Ready");
}

void loop() {
  handleSerial();
  monitorTimeout();
}

void handleSerial() {
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\\n');
    cmd.trim();
    if (cmd.startsWith("S,")) {
      int duration = cmd.substring(2).toInt();
      if (duration > 0 && duration <= SPRAY_TIMEOUT) {
        triggerSpray(duration);
      } else {
        Serial.println("⚠️ Invalid duration");
      }
    }
  }
}

void triggerSpray(int durationMs) {
  sprayStart = millis();
  spraying = true;
  sprayCount++;

  Serial.print("🔄 SPRAY #");
  Serial.print(sprayCount);
  Serial.print(" | Duration: ");
  Serial.println(durationMs);

  analogWrite(SPRAY_PIN, 200); // PWM ~80%
  digitalWrite(LED_PIN, HIGH);
  delay(durationMs);
  analogWrite(SPRAY_PIN, 0);
  digitalWrite(LED_PIN, LOW);
  spraying = false;

  float flow = analogRead(FLOW_PIN) * (5.0 / 1023.0);
  Serial.print("✅ Flow voltage: ");
  Serial.println(flow, 2);
}

void monitorTimeout() {
  if (spraying && millis() - sprayStart > SPRAY_TIMEOUT) {
    Serial.println("🛑 SPRAY TIMEOUT");
    analogWrite(SPRAY_PIN, 0);
    digitalWrite(LED_PIN, LOW);
    spraying = false;
  }
}
"""

# Step 5: Electrostatic sprayer + gimbal control (PWM + servo)

electrostatic_gimbal_ino = """
// electrostatic_gimbal_sprayer.ino – Controls PWM sprayer + servo gimbal nozzle

#include <Servo.h>

#define PWM_SPRAY_PIN 9       // PWM pin to activate electrostatic nozzle
#define GIMBAL_SERVO_PIN 10   // Servo pin for aiming
#define LED_PIN 13            // Status LED
#define FLOW_PIN A0           // Optional feedback

#define SPRAY_TIMEOUT 5000    // Safety cap in ms

Servo gimbal;
unsigned long sprayStart = 0;
bool spraying = false;

void setup() {
  pinMode(LED_PIN, OUTPUT);
  pinMode(PWM_SPRAY_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  analogWrite(PWM_SPRAY_PIN, 0);

  gimbal.attach(GIMBAL_SERVO_PIN);
  gimbal.write(90);  // Neutral position

  Serial.begin(9600);
  while (!Serial);
  Serial.println("🟢 Electrostatic Sprayer + Gimbal Ready");
}

void loop() {
  handleCommand();
  monitorTimeout();
}

void handleCommand() {
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\\n');
    cmd.trim();

    if (cmd.startsWith("S,")) {
      int duration = cmd.substring(2).toInt();
      if (duration > 0 && duration <= SPRAY_TIMEOUT) {
        spray(duration);
      }
    } else if (cmd.startsWith("A,")) {
      int angle = cmd.substring(2).toInt();
      gimbal.write(constrain(angle, 45, 135));  // Limit range
      Serial.print("🎯 Gimbal angle set to ");
      Serial.println(angle);
    }
  }
}

void spray(int durationMs) {
  sprayStart = millis();
  spraying = true;
  Serial.print("💨 Spraying for ");
  Serial.print(durationMs);
  Serial.println(" ms");

  analogWrite(PWM_SPRAY_PIN, 200);  // ~80% duty
  digitalWrite(LED_PIN, HIGH);
  delay(durationMs);
  analogWrite(PWM_SPRAY_PIN, 0);
  digitalWrite(LED_PIN, LOW);
  spraying = false;

  float flow = analogRead(FLOW_PIN) * (5.0 / 1023.0);
  Serial.print("✅ Flow voltage: ");
  Serial.println(flow, 2);
}

void monitorTimeout() {
  if (spraying && (millis() - sprayStart > SPRAY_TIMEOUT)) {
    analogWrite(PWM_SPRAY_PIN, 0);
    digitalWrite(LED_PIN, LOW);
    spraying = false;
    Serial.println("🛑 SPRAY TIMEOUT");
  }
}
"""

wind_compensator_fixed = """
# wind_compensator.py – Adjust spray parameters based on wind direction and speed

import math

def compensate_spray(wind_speed, wind_direction_deg, tree_direction_deg, base_duration):
    \"\"\"
    Adjust spray duration and gimbal angle based on relative wind

    Args:
        wind_speed (float): wind in m/s
        wind_direction_deg (float): wind heading (0–360°, from where it comes)
        tree_direction_deg (float): drone-to-tree bearing (0–360°)
        base_duration (int): base spray duration in ms

    Returns:
        dict: {'duration_ms': int, 'gimbal_angle': int}
    \"\"\"
    relative_angle = (tree_direction_deg - wind_direction_deg + 360) % 360
    angle_offset = 90 - abs(180 - relative_angle)  # closer to 90 = more crosswind

    # Spray longer in headwind, less in tailwind, and adjust angle in crosswind
    if relative_angle < 90 or relative_angle > 270:
        duration = int(base_duration * 1.15)  # headwind: more time
    elif 90 < relative_angle < 270:
        duration = int(base_duration * 0.85)  # tailwind: less time
    else:
        duration = base_duration

    # Map crosswind to gimbal angle offset
    gimbal_angle = 90 + int((angle_offset / 90) * 20) * (-1 if relative_angle < 180 else 1)

    return {
        "duration_ms": max(500, min(duration, 5000)),
        "gimbal_angle": max(60, min(gimbal_angle, 120))
    }
"""

# Step 7: Retry queue logic integrated into px4_spray_sync

retry_queue_logic = """
# retry_logic_spray.py – Add retry queue for missed spray attempts

import pandas as pd
from geopy.distance import geodesic
import serial
import asyncio
from mavsdk import System

spray_df = pd.read_csv("spray_commands_linked.csv")
sprayed = set()
retry_queue = []
ser = serial.Serial("/dev/ttyUSB0", 9600, timeout=1)

def match_gps(lat, lon, radius=4.0):
    for _, row in spray_df.iterrows():
        if row.tree_id in sprayed:
            continue
        dist = geodesic((lat, lon), (row.lat, row.lon)).meters
        if dist < radius:
            return row
    return None

async def main():
    drone = System()
    await drone.connect(system_address="serial:///dev/ttyACM0:57600")
    async for pos in drone.telemetry.position():
        match = match_gps(pos.latitude_deg, pos.longitude_deg)
        if match:
            try:
                print(f"🛰️ At {match.tree_id}: Spraying {match.command}")
                ser.write((match.command + "\\n").encode())
                sprayed.add(match.tree_id)
                await asyncio.sleep(3)
            except Exception as e:
                print(f"⚠️ Spray failed: {e}. Queuing for retry.")
                retry_queue.append(match)

        # Retry logic (1 per loop)
        if retry_queue:
            retry = retry_queue[0]
            try:
                print(f"🔁 Retrying {retry.tree_id}")
                ser.write((retry.command + "\\n").encode())
                sprayed.add(retry.tree_id)
                retry_queue.pop(0)
                await asyncio.sleep(3)
            except:
                print("⚠️ Retry failed. Will retry later.")
"""

path = "/mnt/data/retry_logic_spray.py"
with open(path, "w") as f:
    f.write(retry_queue_logic.strip())

# battery_route_splitter.py – Split long route into battery-safe segments

def split_waypoints(waypoints, max_per_segment=30):
    \"\"\"
    Splits waypoints into chunks for multiple safe flight segments

    Args:
        waypoints (list): list of waypoints
        max_per_segment (int): max points per mission (battery safe)

    Returns:
        List of lists (segments)
    \"\"\"
    return [waypoints[i:i+max_per_segment] for i in range(0, len(waypoints), max_per_segment)]

def generate_segmented_plans(waypoints, base_filename="segment", altitude=10):
    import json
    segments = split_waypoints(waypoints)

    for idx, seg in enumerate(segments):
        plan = {
            "fileType": "Plan",

            "rallyPoints": {"points": [], "version": 2},

tree_proximity_checker = """
# tree_proximity_checker.py – Simple LiDAR-based tree collision avoidance

import time
import serial

# Assume LiDAR (e.g., TFmini-S) connected to /dev/ttyUSB1
# Returns distance in cm

def get_lidar_distance(port="/dev/ttyUSB1"):
    try:
        with serial.Serial(port, 115200, timeout=1) as ser:
            data = ser.read(9)
            if data[0] == 0x59 and data[1] == 0x59:
                distance = data[2] + data[3]*256
                return distance
    except:
        return -1
    return -1

def check_proximity(threshold_cm=200):
    distance = get_lidar_distance()
    if 0 < distance < threshold_cm:
        print(f"🚨 Obstacle Detected at {distance} cm!")
        return True
    return False

if __name__ == "__main__":
    while True:
        if check_proximity():
            print("🛑 HALT or ADJUST TRAJECTORY")
        time.sleep(0.2)
"""
